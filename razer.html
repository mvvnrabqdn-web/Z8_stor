from flask import Flask, request, jsonify, render_template_string from selenium import webdriver from selenium.webdriver.common.by import By from selenium.webdriver.common.keys import Keys from selenium.webdriver.support.ui import WebDriverWait from selenium.webdriver.support import expected_conditions as EC from selenium.webdriver.chrome.service import Service from selenium.webdriver.chrome.options import Options from webdriver_manager.chrome import ChromeDriverManager import threading import time import random import os app = Flask(__name__) # ===== متغيرات التحكم ===== bot_status = "متوقف" bot_thread = None stop_flag = False # بيانات الدخول ونص التعليق username = "" password = "" comment_text = "" target_hashtag = "photography" # الهاشتاغ الافتراضي # ===== واجهة HTML (جميلة ومدمجة) ===== HTML_PAGE = """ بوت إنستغرام الذكي

⚡ ذكاء التعليقات

اكتب تعليقك وابدأ الهجوم الذكي

📡 الحالة: متوقف
⚠️ سيفتح متصفح Chrome أمامك. لا تغلقه أثناء التشغيل. ⚠️
""" # ===== منطق البوت (Selenium + تعليق مخصص) ===== def run_bot(): global bot_status, stop_flag, username, password, comment_text, target_hashtag driver = None try: bot_status = "🔄 جاري تهيئة المتصفح..." # إعدادات Chrome لتجنب الكشف options = Options() options.add_argument("--start-maximized") options.add_argument("--disable-blink-features=AutomationControlled") options.add_experimental_option("excludeSwitches", ["enable-automation"]) options.add_experimental_option('useAutomationExtension', False) # تشغيل المتصفح service = Service(ChromeDriverManager().install()) driver = webdriver.Chrome(service=service, options=options) driver.execute_script("Object.defineProperty(navigator, 'webdriver', {get: () => undefined})") # ===== 1. تسجيل الدخول ===== bot_status = "🔓 جاري تسجيل الدخول..." driver.get("https://www.instagram.com/accounts/login/") time.sleep(3) # إدخال البيانات username_input = WebDriverWait(driver, 15).until( EC.presence_of_element_located((By.NAME, "username")) ) username_input.send_keys(username) driver.find_element(By.NAME, "password").send_keys(password) driver.find_element(By.NAME, "password").send_keys(Keys.RETURN) time.sleep(5) # تخطي "حفظ المعلومات" إذا ظهر try: not_now = WebDriverWait(driver, 5).until( EC.element_to_be_clickable((By.XPATH, "//div[contains(text(),'ليس الآن')]")) ) not_now.click() time.sleep(2) except: pass # ===== 2. الدخول للريلزات ===== bot_status = f"🔍 جاري البحث عن ريلزات #{ target_hashtag}..." driver.get(f"https://www.instagram.com/explore/tags/{target_hashtag}/") time.sleep(4) # النقر على أول ريلز try: first_reel = WebDriverWait(driver, 15).until( EC.element_to_be_clickable((By.XPATH, "//div[@class='_aagv']//a[contains(@href, '/reel/')]")) ) first_reel.click() time.sleep(4) except Exception as e: bot_status = f"⚠️ لم أجد ريلزات للهاشتاغ #{ target_hashtag}" return # ===== 3. حلقة التعليق والإعجاب ===== counter = 0 while not stop_flag: counter += 1 bot_status = f"🔄 جولة {counter} - جاري التفاعل..." try: # --- إعجاب (Like) --- try: like_btn = WebDriverWait(driver, 5).until( EC.element_to_be_clickable((By.XPATH, "//span[@class='x1rg5ohu']//*[name()='svg' and @aria-label='إعجاب']")) ) like_btn.click() time.sleep(random.uniform(1, 3)) bot_status = f"❤️ جولة {counter}: تم الإعجاب!" except: bot_status = f"⚠️ جولة {counter}: زر الإعجاب غير موجود (قد يكون معجب مسبقاً)" # --- تعليق (Comment) --- try: # النقر على حقل التعليق comment_field = WebDriverWait(driver, 10).until( EC.element_to_be_clickable((By.XPATH, "//textarea[@placeholder='أضف تعليقًا...']")) ) comment_field.click() time.sleep(1) # كتابة النص المخصص (الذي كتبته أنت) comment_field.send_keys(comment_text) time.sleep(1) # الضغط على زر النشر post_btn = driver.find_element(By.XPATH, "//div[contains(text(),'نشر')]") post_btn.click() bot_status = f"💬 جولة {counter}: تم التعليق: '{ comment_text[:20]}...'" time.sleep(random.uniform(2, 4)) except Exception as e: bot_status = f"⚠️ جولة {counter}: فشل التعليق ({str(e)[:30]})" # ===== الانتقال للريلز التالي ===== driver.find_element(By.TAG_NAME, 'body').send_keys(Keys.ARROW_DOWN) time.sleep(random.uniform(3, 5)) # انتظار طويل نسبياً عشان ما ينحظر الحساب wait_time = random.randint(12, 25) bot_status = f"⏳ انتظار {wait_time} ثانية قبل الجولة القادمة..." for _ in range(wait_time): if stop_flag: break time.sleep(1) except Exception as e: bot_status = f"❌ خطأ في الجولة {counter}: {str(e)[:50]}" time.sleep(5) # محاولة العودة للريلزات try: driver.get(f"https://www.instagram.com/explore/tags/{target_hashtag}/") time.sleep(4) first_reel = WebDriverWait(driver, 10).until( EC.element_to_be_clickable((By.XPATH, "//div[@class='_aagv']//a[contains(@href, '/reel/')]")) ) first_reel.click() time.sleep(4) except: pass bot_status = "🛑 تم الإيقاف بواسطتك" except Exception as e: bot_status = f"💀 خطأ فادح: {str(e)[:80]}" print("="*50) print("الخطأ الكامل:") import traceback traceback.print_exc() print("="*50) finally: if driver: driver.quit() # ===== مسارات الموقع ===== @app.route('/') def index(): return render_template_string(HTML_PAGE) @app.route('/start', methods=['POST']) def start(): global bot_thread, stop_flag, bot_status, username, password, comment_text, target_hashtag if bot_thread and bot_thread.is_alive(): return jsonify({"status": "⚠️ البوت يعمل بالفعل!"}) data = request.json username = data.get('username') password = data.get('password') target_hashtag = data.get('hashtag', 'photography') comment_text = data.get('comment', '🔥 ريلز رائع!') if not username or not password: return jsonify({"status": "❌ يرجى إدخال اسم المستخدم وكلمة المرور!"}) stop_flag = False bot_status = "🚀 جاري الإقلاع..." bot_thread = threading.Thread(target=run_bot) bot_thread.start() return jsonify({"status": f"✅ تم التشغيل! سيتفاعل مع #{ target_hashtag}"}) @app.route('/stop', methods=['POST']) def stop(): global stop_flag, bot_status stop_flag = True bot_status = "⏳ جاري الإيقاف..." return jsonify({"status": "⏹ تم إرسال أمر الإيقاف"}) @app.route('/status') def status(): return jsonify({"status": bot_status}) # ===== الإقلاع ===== if __name__ == '__main__': print("="*50) print("🔥 Shadow Bot V3 - النسخة الذكية") print("📌 افتح المتصفح على: http://127.0.0.1:5000") print("📌 اكتب تعليقك المخصص، وابدأ الهجوم الذكي!") print("⚠️ تأكد أن Google Chrome مثبت على جهازك.") print("="*50) app.run(debug=True, host='0.0.0.0', port=5000)
