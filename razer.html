from flask import Flask, request, jsonify, render_template_string from instapy import InstaPy from instapy import smart_run import threading import time app = Flask(__name__) # ===== متغيرات التحكم ===== bot_status = "متوقف" bot_thread = None stop_flag = False username = "" password = "" # ===== واجهة HTML مدمجة ===== HTML_PAGE = """ بوت إنستغرام - لوحة تحكم

⚡ نينجا التعليقات

📡 الحالة: متوقف
🔥 يعلق تلقائياً على ريلزات #photography و #nature 🔥
""" # ===== وظائف البوت ===== def run_bot(): global bot_status, stop_flag, username, password try: bot_status = "جاري التشغيل..." session = InstaPy(username=username, password=password, headless_browser=False) with smart_run(session): session.set_do_comment(True, percentage=100) session.set_comments([ "🔥 ريلز رائع!", "❤️ تحفة فنية", "😍 إبداع", "🚀 يستحق المشاهدة", "💯 محتوى قيم" ]) while not stop_flag: session.like_by_tags(["photography", "nature"], amount=5) time.sleep(15) bot_status = "متوقف (تم الإنهاء)" except Exception as e: bot_status = f"خطأ: {str(e)[:50]}" # ===== مسارات الموقع ===== @app.route('/') def index(): return render_template_string(HTML_PAGE) @app.route('/start', methods=['POST']) def start(): global bot_thread, stop_flag, bot_status, username, password if bot_thread and bot_thread.is_alive(): return jsonify({"status": "⚠️ البوت يعمل بالفعل!"}) data = request.json username = data.get('username') password = data.get('password') stop_flag = False bot_thread = threading.Thread(target=run_bot) bot_thread.start() return jsonify({"status": "✅ تم تشغيل البوت!"}) @app.route('/stop', methods=['POST']) def stop(): global stop_flag, bot_status stop_flag = True bot_status = "⏳ جاري الإيقاف..." return jsonify({"status": "⏹ تم إرسال أمر الإيقاف"}) @app.route('/status') def status(): return jsonify({"status": bot_status}) # ===== الإقلاع ===== if __name__ == '__main__': print("🔥 Shadow Bot جاهز! افتح المتصفح على: http://127.0.0.1:5000") app.run(debug=True, host='0.0.0.0', port=5000)
