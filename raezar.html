const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); app.post("/chat", async (req, res) => { const userMessage = req.body.message; // رد ذكي بسيط (تقدر تربطه AI حقيقي بعدين) let reply; if (userMessage.includes("مرحبا")) { reply = "هلا بيك 👋"; } else if (userMessage.includes("شلونك")) { reply = "تمام وانت؟"; } else { reply = "ما فهمت عليك بس احاول اتعلم 🤖"; } res.json({ reply }); }); app.listen(3000, () => { console.log("Server شغال على http://localhost:3000"); });
